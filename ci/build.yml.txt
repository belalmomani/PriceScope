# Move this file to .github/workflows/build.yml (rename it in the GitHub web editor).
name: Build

on:
  push:
    branches: [main]
  workflow_dispatch:
    inputs:
      release:
        description: "Build a SIGNED release (APK + AAB) and tag it"
        type: boolean
        default: false

concurrency:
  group: build-${{ github.ref }}
  cancel-in-progress: false

env:
  GRADLE_VERSION: "8.13"

jobs:
  test-and-debug:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: "17"
      - uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: ${{ env.GRADLE_VERSION }}
      - name: Install Android platform
        run: |
          yes | "$ANDROID_HOME/cmdline-tools/latest/bin/sdkmanager" --licenses > /dev/null || true
          "$ANDROID_HOME/cmdline-tools/latest/bin/sdkmanager" "platforms;android-36"
      - name: Unit tests and debug APK
        run: gradle --no-daemon testDebugUnitTest assembleDebug
      - uses: actions/upload-artifact@v4
        with:
          name: pricescope-debug-apk
          path: app/build/outputs/apk/debug/*.apk
          if-no-files-found: error
      - uses: actions/upload-artifact@v4
        if: always()
        with:
          name: test-reports
          path: app/build/reports/tests/
          if-no-files-found: ignore

  release:
    if: ${{ github.event_name == 'workflow_dispatch' && inputs.release }}
    needs: test-and-debug
    runs-on: ubuntu-latest
    permissions:
      contents: write
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0
      - uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: "17"
      - uses: gradle/actions/setup-gradle@v4
        with:
          gradle-version: ${{ env.GRADLE_VERSION }}
      - name: Install Android platform
        run: |
          yes | "$ANDROID_HOME/cmdline-tools/latest/bin/sdkmanager" --licenses > /dev/null || true
          "$ANDROID_HOME/cmdline-tools/latest/bin/sdkmanager" "platforms;android-36"

      - name: Version guard (versionCode must always increase)
        run: |
          VERSION_CODE=$(grep '^VERSION_CODE=' version.properties | cut -d= -f2 | tr -d '[:space:]')
          VERSION_NAME=$(grep '^VERSION_NAME=' version.properties | cut -d= -f2 | tr -d '[:space:]')
          MAX=0
          for t in $(git tag --list 'v*'); do
            c=$(git show "$t:version.properties" 2>/dev/null | grep '^VERSION_CODE=' | cut -d= -f2 | tr -d '[:space:]' || true)
            if [ -n "$c" ] && [ "$c" -gt "$MAX" ]; then MAX=$c; fi
          done
          echo "Current versionCode=$VERSION_CODE, highest released=$MAX"
          if [ "$VERSION_CODE" -le "$MAX" ]; then
            echo "::error::VERSION_CODE ($VERSION_CODE) must be greater than the last release ($MAX). Edit version.properties."
            exit 1
          fi
          if git rev-parse "v$VERSION_NAME" >/dev/null 2>&1; then
            echo "::error::Tag v$VERSION_NAME already exists. Increase VERSION_NAME and VERSION_CODE."
            exit 1
          fi
          echo "VERSION_NAME=$VERSION_NAME" >> "$GITHUB_ENV"

      - name: Restore the existing signing key from secrets
        env:
          KEYSTORE_BASE64: ${{ secrets.KEYSTORE_BASE64 }}
          KEYSTORE_PASSWORD: ${{ secrets.KEYSTORE_PASSWORD }}
          KEY_ALIAS: ${{ secrets.KEY_ALIAS }}
          KEY_PASSWORD: ${{ secrets.KEY_PASSWORD }}
        run: |
          for v in KEYSTORE_BASE64 KEYSTORE_PASSWORD KEY_ALIAS KEY_PASSWORD; do
            if [ -z "${!v}" ]; then
              echo "::error::Missing GitHub secret $v. No new key is ever generated automatically."
              exit 1
            fi
          done
          echo "$KEYSTORE_BASE64" | base64 -d > "$RUNNER_TEMP/release.jks"
          echo "KEYSTORE_PATH=$RUNNER_TEMP/release.jks" >> "$GITHUB_ENV"
          {
            echo "### Signing certificate fingerprints (add these to Firebase)"
            keytool -list -v -keystore "$RUNNER_TEMP/release.jks" -storepass "$KEYSTORE_PASSWORD" -alias "$KEY_ALIAS" \
              | grep -E "SHA1:|SHA256:" | sed 's/^ *//'
          } >> "$GITHUB_STEP_SUMMARY"

      - name: Build signed release
        env:
          KEYSTORE_PASSWORD: ${{ secrets.KEYSTORE_PASSWORD }}
          KEY_ALIAS: ${{ secrets.KEY_ALIAS }}
          KEY_PASSWORD: ${{ secrets.KEY_PASSWORD }}
        run: gradle --no-daemon bundleRelease assembleRelease

      - name: Verify APK signature
        run: |
          APKSIGNER=$(ls "$ANDROID_HOME"/build-tools/*/apksigner | sort -V | tail -1)
          "$APKSIGNER" verify --print-certs app/build/outputs/apk/release/*.apk | tee -a "$GITHUB_STEP_SUMMARY"

      - uses: actions/upload-artifact@v4
        with:
          name: pricescope-release-${{ env.VERSION_NAME }}
          path: |
            app/build/outputs/apk/release/*.apk
            app/build/outputs/bundle/release/*.aab
          if-no-files-found: error

      - name: Tag the release
        run: |
          git config user.name "github-actions"
          git config user.email "github-actions@users.noreply.github.com"
          git tag "v$VERSION_NAME"
          git push origin "v$VERSION_NAME"
